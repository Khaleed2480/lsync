Link extension via POST /api/extension/link using JWT loginMsg.textContent = 'Linking extension...'; const linkRes = await fetch(`${API_BASE}/extension/link`, { method: 'POST', headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${jwt}`, }, }); const linkData = await linkRes.json().catch(() => ({})); console.log('[LSync] /extension/link response:', linkData); if (!linkRes.ok || linkData.success === false) { throw new Error(linkData.message || 'Could not link extension.'); } // Check all common token property names returned by backend const extToken = linkData.extensionToken || linkData.token || linkData.key || linkData.apiKey || linkData.secret || linkData.data?.extensionToken || linkData.data?.token || linkData.data?.key; fetchRecordedEvents(); } catch (err) { loginMsg.style.color = '#F87171'; loginMsg.textContent = err.message || 'Connection failed.'; } finally { loginBtn.disabled = false; } }); // ================= STEP 3: MANUAL SYNC CURRENT YOUTUBE VIDEO ================= syncNowBtn.addEventListener('click', async () => { syncMsg.style.color = '#F4A836'; } else { syncMsg.style.color = '#F87171'; syncMsg.textContent = response.error || 'Sync failed.'; } }); }); // ================= STEP 4: LIST RECORDED EVENTS ================= async function fetchRecordedEvents() { eventsList.innerHTML = '
Loading...
'; const keys = await chrome.storage.local.get(['extensionToken', 'loginToken']); const authToken = keys.extensionToken || keys.loginToken; try { let res = await fetch(`${API_BASE}/watch-events?page=1&limit=20`, { method: 'GET', headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${authToken}`, 'x-extension-token': authToken, }, }); if (res.status === 401 && keys.loginToken && authToken !== keys.loginToken) { res = await fetch(`${API_BASE}/watch-events?page=1&limit=20`, { method: 'GET', headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${keys.loginToken}`, }, }); } const data = await res.json().catch(() => ({})); const items = Array.isArray(data) ? data : data.events || data.watchEvents || data.data || []; if (items.length === 0) { eventsList.innerHTML = '

'; } } refreshEvents.addEventListener('click', fetchRecordedEvents); // Disconnect / Sign Out logoutBtn.addEventListener('click', async () => { await chrome.storage.local.remove(['extensionToken', 'loginToken', 'userName']); showLoginView(); loginMsg.textContent = ''; syncMsg.textContent = ''; }); });
